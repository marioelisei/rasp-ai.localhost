Prepare the System

~15 min

Use Raspberry Pi OS Lite 64-bit (no desktop — saves RAM). Flash with Raspberry Pi Imager.

terminal — system update 
# Update everything first
sudo apt update && sudo apt upgrade -y

# Install audio + Python dependencies
sudo apt install -y python3-pip python3-venv ffmpeg \
  portaudio19-dev libasound2-dev pulseaudio \
  alsa-utils git curl

# Verify microphone is detected
arecord -l

# Verify speaker is detected
aplay -l
ℹ
Audio tip If the USB mic doesn't appear, run sudo raspi-config → System Options → Audio and select the correct device.

Install Ollama + LLM

~20 min

Ollama runs LLMs locally via a REST API. Native ARM64 support on Pi 5.

terminal — install Ollama 
# Install Ollama (native ARM64)
curl -fsSL https://ollama.com/install.sh | sh

# Start the service in background
ollama serve &

# Pull a model (see table below for options)

Integration Script — Full Pipeline

core of the project

This script connects the three components in a continuous loop: record → transcribe → reason → speak.

~/assistant/main.py 
import sounddevice as sd
import numpy as np
import requests
import subprocess
import tempfile
import os
from faster_whisper import WhisperModel

# ── Configuration ──────────────────────────────────────
WHISPER_MODEL  = "small"         # tiny | base | small | medium
OLLAMA_MODEL   = "gemma3:4b"     # any model pulled in Ollama
VOICE_MODEL    = "/root/voices/pt_BR-faber-medium.onnx"
OLLAMA_URL     = "http://localhost:11434/api/generate"
SAMPLERATE     = 16000
RECORD_SECONDS = 5

# ── Load Whisper once ───────────────────────────────────
print("⏳ Loading Whisper...")
whisper = WhisperModel(WHISPER_MODEL, device="cpu", compute_type="int8")
print("✅ Whisper ready")

# ── Conversation history ────────────────────────────────
history = []

def record_audio():
    print("\n🎙 Speak now...")
    audio = sd.rec(
        int(RECORD_SECONDS * SAMPLERATE),
        samplerate=SAMPLERATE, channels=1, dtype="float32"
    )
    sd.wait()
    return audio.flatten()

def transcribe(audio):
    segments, _ = whisper.transcribe(audio, language="pt")
    text = " ".join(s.text for s in segments).strip()
    print(f"🗣 You said: {text}")
    return text

def ask_llm(text):
    history.append({"role": "user", "content": text})
    prompt = "\n".join(
        f"{'User' if m['role']=='user' else 'Assistant'}: {m['content']}"
        for m in history[-6:]
    ) + "\nAssistant:"

    resp = requests.post(OLLAMA_URL, json={
        "model": OLLAMA_MODEL,
        "prompt": prompt,
        "stream": False,
        "options": {"temperature": 0.7, "num_predict": 200}
    })
    answer = resp.json()["response"].strip()
    history.append({"role": "assistant", "content": answer})
    print(f"🤖 Assistant: {answer}")
    return answer

def speak(text):
    with tempfile.NamedTemporaryFile(suffix=".wav", delete=False) as f:
        wav = f.name
    subprocess.run(
        ["python3", "-m", "piper", "--model", VOICE_MODEL, "--output_file", wav],
        input=text.encode(), capture_output=True
    )
    subprocess.run(["aplay", wav])
    os.unlink(wav)

# ── Main loop ───────────────────────────────────────────
print("\n🟢 Offline assistant started! Ctrl+C to exit.\n")
while True:
    try:
        audio  = record_audio()
        text   = transcribe(audio)
        if not text:
            continue
        answer = ask_llm(text)
        speak(answer)
    except KeyboardInterrupt:
        print("\n👋 Shutting down...")
        break

python3 ~/assistant/main.py

Auto-start on Boot

optional

Create a systemd service so the assistant starts automatically when the Pi powers on.

/etc/systemd/system/assistant.service 
[Unit]
Description=Offline AI Voice Assistant
After=network.target ollama.service

[Service]
User=pi
WorkingDirectory=/home/pi
ExecStart=/home/pi/assistant/bin/python3 /home/pi/assistant/main.py
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
terminal 
sudo systemctl enable assistant
sudo systemctl start assistant
sudo systemctl status assistant
